
import os
import cv2
import json
import numpy as np
import mediapipe as mp
import glob
from tensorflow.keras.models import load_model

# ==========================================
# 1. PATHS & CONFIGURATION
# ==========================================
MODEL_PATH = "/kaggle/input/datasets/amandeep233/final-2731-model/best_model_2731.keras"
DECODER_JSON = "/kaggle/input/datasets/amandeep233/final-2731-model/index_to_gloss_2731.json"
TEST_DIR = "/kaggle/input/datasets/amandeep233/citizen-vid-test-2-0"

# ==========================================
# 2. TOBY'S EXACT PREPROCESSING LOGIC
# ==========================================
def hand_normalize(hand_data):
    xmin, xmax = np.min(hand_data[:,0]), np.max(hand_data[:,0])
    ymin, ymax = np.min(hand_data[:,1]), np.max(hand_data[:,1])
    width, height = xmax - xmin, ymax - ymin
    center = np.array([(xmin+xmax)/2, (ymin+ymax)/2])
    
    if width != 0: hand_data[:,0] = (hand_data[:,0]-center[0])/width
    else: hand_data[:,0] = 0
    if height != 0: hand_data[:,1] = (hand_data[:,1]-center[1])/height
    else: hand_data[:,1] = 0
    return hand_data

def distance(x1, x2): 
    delta = x1 - x2
    return (delta[0]**2 + delta[1]**2)**0.5

def AnchorNorm(target, scale, reference):
    return (target - reference) / (scale + 0.01)

def pad_video(X):
    # Toby loops the video frames until it is exactly 150 frames long
    padded_length = 150
    data_array = np.zeros((padded_length, X.shape[1], X.shape[2]))
    for landmark in range(0, X.shape[1]):
        for coordinate in range(0, X.shape[2]):
            data_array[:, landmark, coordinate] = np.tile(X[:, landmark, coordinate], int(padded_length/X.shape[0]+2))[:padded_length]
    return data_array

def extract_landmarks(results):
    important_face_landmarks = [
        33, 133, 159, 145, 153, 144, 362, 263, 386, 374, 380, 373,
        70, 63, 105, 66, 107, 295, 282, 320, 285, 318,
        1, 168, 197, 4, 78, 308, 13, 14, 81, 311
    ]
    
    # Strip lower body and head pose, keeping exactly 12 pose points
    pose_indices_to_remove = set(list(range(23, 33)) + list(range(0, 11)))
    pose_indices_to_keep = [i for i in range(33) if i not in pose_indices_to_remove]

    pose = [(lm.x, lm.y) for i, lm in enumerate(results.pose_landmarks.landmark) if i in pose_indices_to_keep] if results.pose_landmarks else [(0, 0)] * len(pose_indices_to_keep)
    face = [(results.face_landmarks.landmark[i].x, results.face_landmarks.landmark[i].y) for i in important_face_landmarks] if results.face_landmarks else [(0, 0)] * len(important_face_landmarks)
    left_hand = [(lm.x, lm.y) for lm in results.left_hand_landmarks.landmark] if results.left_hand_landmarks else [(0, 0)] * 21
    right_hand = [(lm.x, lm.y) for lm in results.right_hand_landmarks.landmark] if results.right_hand_landmarks else [(0, 0)] * 21

    # Exact Order: Pose(12) + LHand(21) + RHand(21) + Face(32) = 86 landmarks
    return np.array(pose + left_hand + right_hand + face)

def preprocess_frame(X):
    # Toby's exact frame-by-frame structural normalization
    neck = abs(X[0] + X[1]) / 2
    X[:] = AnchorNorm(X[:], distance(X[0], X[1]), neck)
    X[54:] = AnchorNorm(X[54:], distance(X[0], X[1]), X[79])
    
    left_arm, right_arm = [2,4,6,8,10], [3,5,7,9,11]
    X[left_arm] = AnchorNorm(X[left_arm], distance(X[0], X[2]), X[0])
    X[right_arm] = AnchorNorm(X[right_arm], distance(X[1], X[3]), X[1])
    
    X[12:33] = hand_normalize(X[12:33])
    X[33:54] = hand_normalize(X[33:54])
    return X

# ==========================================
# 3. EXECUTION SCRIPT
# ==========================================
print("🔄 Loading 2731-Class GRU Model...")
model = load_model(MODEL_PATH)

with open(DECODER_JSON, 'r') as f:
    raw_decoder = json.load(f)
    index_to_gloss = {int(k): v for k, v in raw_decoder.items()}

print("✅ Model Loaded Successfully!\n")

video_files = glob.glob(os.path.join(TEST_DIR, "*.mp4"))

if len(video_files) == 0:
    print(f"❌ No videos found in {TEST_DIR}")
else:
    mp_holistic = mp.solutions.holistic
    
    for video_path in video_files:
        video_name = os.path.basename(video_path)
        cap = cv2.VideoCapture(video_path)
        frames_features = []
        
        with mp_holistic.Holistic(static_image_mode=False, min_detection_confidence=0.5) as holistic:
            while cap.isOpened():
                ret, frame = cap.read()
                if not ret: break
                
                # Toby horizontally flips the frame for camera mirroring during live prep!
                # Since these are pre-recorded videos, we should process them directly
                image_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                results = holistic.process(image_rgb)
                
                landmarks = extract_landmarks(results)
                processed_landmarks = preprocess_frame(landmarks)
                frames_features.append(processed_landmarks)
                
        cap.release()
        
        if len(frames_features) == 0:
            print(f"{video_name} : answer = UNKNOWN (No frames extracted)")
            continue
            
        # 1. Convert to Array -> Shape: (Frames, 86, 2)
        landmark_array = np.array(frames_features, dtype=np.float32)
        
        # 2. Tile to exactly 150 frames -> Shape: (150, 86, 2)
        padded_array = pad_video(landmark_array)
        
        # 3. Flatten the spatial dimensions -> Shape: (150, 172)
        reshape_array = padded_array.reshape(padded_array.shape[0], -1)
        
        # 4. Add batch dimension -> Shape: (1, 150, 172)
        model_input = np.expand_dims(reshape_array, axis=0)
        
        # Predict
        predictions = model.predict(model_input, verbose=0)[0]
        best_pred_idx = np.argmax(predictions)
        predicted_label = index_to_gloss.get(best_pred_idx, "UNKNOWN")
        
        print(f"{video_name} : answer = {predicted_label}")

print("\n🎉 Testing Complete!")
